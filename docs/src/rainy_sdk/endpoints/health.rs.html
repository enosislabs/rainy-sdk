1//! Public health operations.
2
3use crate::client::RainyClient;
4use crate::error::Result;
5use crate::models::{HealthStatus, ServiceStatus};
6use serde::Deserialize;
7use serde_json::Value;
8
9impl RainyClient {
10    /// Performs the public root health check.
11    pub async fn health_check(&self) -> Result<HealthStatus> {
12        #[derive(Deserialize)]
13        struct RootHealthResponse {
14            status: String,
15            timestamp: String,
16        }
17
18        self.wait_for_slot().await;
19        let response = self
20            .send_request(self.root_request(reqwest::Method::GET, "/health"))
21            .await?;
22        let payload: RootHealthResponse = self.handle_response(response).await?;
23        Ok(HealthStatus {
24            status: payload.status,
25            timestamp: payload.timestamp,
26            uptime: 0.0,
27            services: ServiceStatus {
28                database: false,
29                redis: None,
30                providers: false,
31            },
32        })
33    }
34
35    /// Performs a detailed health check when the service exposes one.
36    ///
37    /// Only generic dependency labels are interpreted. Unknown fields are
38    /// ignored so the SDK does not encode service-internal topology.
39    pub async fn detailed_health_check(&self) -> Result<HealthStatus> {
40        #[derive(Deserialize)]
41        struct DependenciesHealthResponse {
42            status: String,
43            timestamp: String,
44            #[serde(default)]
45            dependencies: Value,
46        }
47
48        self.wait_for_slot().await;
49        let response = self
50            .send_request(self.root_request(reqwest::Method::GET, "/health/dependencies"))
51            .await?;
52        let payload: DependenciesHealthResponse = self.handle_response(response).await?;
53        let dependencies = payload.dependencies.as_object();
54        let bool_value = |names: &[&str]| {
55            names
56                .iter()
57                .find_map(|name| dependencies.and_then(|items| items.get(*name)))
58                .and_then(Value::as_bool)
59                .unwrap_or(false)
60        };
61        Ok(HealthStatus {
62            status: payload.status,
63            timestamp: payload.timestamp,
64            uptime: 0.0,
65            services: ServiceStatus {
66                database: bool_value(&["database"]),
67                redis: dependencies
68                    .and_then(|items| items.get("redis"))
69                    .and_then(Value::as_bool),
70                providers: bool_value(&["providers", "inference"]),
71            },
72        })
73    }
74}
