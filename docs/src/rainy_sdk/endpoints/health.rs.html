6impl RainyClient {
7    /// Performs a basic health check on the Rainy API.
8    ///
9    /// This method is useful for quickly verifying that the API is up and running.
10    ///
11    /// # Returns
12    ///
13    /// A `Result` containing a `HealthStatus` struct with basic health information.
14    ///
15    /// # Example
16    ///
17    /// ```rust,no_run
18    /// # use rainy_sdk::RainyClient;
19    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
20    /// let client = RainyClient::with_api_key("user-api-key")?;
21    /// let health = client.health_check().await?;
22    /// println!("API Status: {}", health.status);
23    /// # Ok(())
24    /// # }
25    /// ```
26    pub async fn health_check(&self) -> Result<HealthStatus> {
27        #[derive(Deserialize)]
28        struct RootHealthResponse {
29            status: String,
30            timestamp: String,
31        }
32
33        self.wait_for_slot().await;
34        let response = self
35            .send_request(self.root_request(reqwest::Method::GET, "/health"))
36            .await?;
37        let payload: RootHealthResponse = self.handle_response(response).await?;
38
39        Ok(HealthStatus {
40            status: payload.status,
41            timestamp: payload.timestamp,
42            uptime: 0.0,
43            services: ServiceStatus {
44                database: false,
45                redis: None,
46                providers: false,
47            },
48        })
49    }

51    /// Performs a detailed health check on the Rainy API and its underlying services.
52    ///
53    /// This method provides more in-depth information, including the status of the database,
54    /// Redis, and connections to AI providers.
55    ///
56    /// # Returns
57    ///
58    /// A `Result` containing a `HealthStatus` struct with detailed service status.
59    ///
60    /// # Example
61    ///
62    /// ```rust,no_run
63    /// # use rainy_sdk::RainyClient;
64    /// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
65    /// let client = RainyClient::with_api_key("user-api-key")?;
66    /// let health = client.detailed_health_check().await?;
67    /// println!("Database status: {}", health.services.database);
68    /// println!("Providers status: {}", health.services.providers);
69    /// # Ok(())
70    /// # }
71    /// ```
72    pub async fn detailed_health_check(&self) -> Result<HealthStatus> {
73        #[derive(Deserialize)]
74        struct DependencyFlags {
75            database: bool,
76            redis: bool,
77            #[serde(rename = "openrouterConfigured")]
78            openrouter_configured: bool,
79            #[serde(rename = "polarConfigured")]
80            polar_configured: bool,
81        }
82        #[derive(Deserialize)]
83        struct DependenciesHealthResponse {
84            status: String,
85            timestamp: String,
86            dependencies: DependencyFlags,
87        }
88
89        self.wait_for_slot().await;
90        let response = self
91            .send_request(self.root_request(reqwest::Method::GET, "/health/dependencies"))
92            .await?;
93        let payload: DependenciesHealthResponse = self.handle_response(response).await?;
94
95        Ok(HealthStatus {
96            status: payload.status,
97            timestamp: payload.timestamp,
98            uptime: 0.0,
99            services: ServiceStatus {
100                database: payload.dependencies.database,
101                redis: Some(payload.dependencies.redis),
102                providers: payload.dependencies.openrouter_configured
103                    && payload.dependencies.polar_configured,
104            },
105        })
106    }
