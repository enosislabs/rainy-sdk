3use crate::client::RainyClient;
4use crate::error::{RainyError, Result};
5use crate::models::{
6    AnthropicMessageRequest, AnthropicMessageResponse, AnthropicMessageStreamEvent,
7};
8use futures::Stream;
9use reqwest::Method;
10use std::pin::Pin;
11
12/// Anthropic's current stable Messages wire version used when callers do not
13/// provide an explicit version.
14pub const DEFAULT_ANTHROPIC_VERSION: &str = "2023-06-01";
15
16fn validate_header(value: &str, name: &str) -> Result<()> {
17    if value.trim().is_empty() {
18        return Err(RainyError::InvalidRequest {
19            code: format!("INVALID_{}", name.to_ascii_uppercase().replace('-', "_")),
20            message: format!("{name} header cannot be empty"),
21            details: None,
22        });
23    }
24    reqwest::header::HeaderValue::from_str(value).map_err(|error| RainyError::InvalidRequest {
25        code: format!("INVALID_{}", name.to_ascii_uppercase().replace('-', "_")),
26        message: format!("{name} header contains invalid characters"),
27        details: Some(serde_json::json!({"error": error.to_string()})),
28    })?;
29    Ok(())
30}
31
32impl RainyClient {
33    /// Creates an Anthropic-compatible Messages response.
34    ///
35    /// The SDK sends the required `anthropic-version: 2023-06-01` header and
36    /// uses the same API-key authentication and bounded response handling as
37    /// the Chat and Responses endpoints.
38    pub async fn create_message(
39        &self,
40        request: AnthropicMessageRequest,
41    ) -> Result<AnthropicMessageResponse> {
42        self.create_message_with_headers(request, DEFAULT_ANTHROPIC_VERSION, None)
43            .await
44    }
45
46    /// Alias for [`Self::create_message`] that makes the provider contract
47    /// explicit at call sites.
48    pub async fn create_anthropic_message(
49        &self,
50        request: AnthropicMessageRequest,
51    ) -> Result<AnthropicMessageResponse> {
52        self.create_message(request).await
53    }
54
55    /// Creates a Messages response with explicit Anthropic version and beta
56    /// headers.
57    pub async fn create_message_with_headers(
58        &self,
59        request: AnthropicMessageRequest,
60        anthropic_version: &str,
61        anthropic_beta: Option<&str>,
62    ) -> Result<AnthropicMessageResponse> {
63        validate_header(anthropic_version, "anthropic-version")?;
64        if let Some(beta) = anthropic_beta {
65            validate_header(beta, "anthropic-beta")?;
66        }
67
68        self.wait_for_slot().await;
69        let mut builder = self
70            .json_request(Method::POST, "/messages", &request)?
71            .header("anthropic-version", anthropic_version);
72        if let Some(beta) = anthropic_beta {
73            builder = builder.header("anthropic-beta", beta);
74        }
75        let response = self.send_request(builder).await?;
76        self.handle_response(response).await
77    }
78
79    /// Creates a streaming Anthropic Messages response using the default wire
80    /// version.
81    pub async fn create_message_stream(
82        &self,
83        mut request: AnthropicMessageRequest,
84    ) -> Result<Pin<Box<dyn Stream<Item = Result<AnthropicMessageStreamEvent>> + Send>>> {
85        request.stream = Some(true);
86        self.create_message_stream_with_headers(request, DEFAULT_ANTHROPIC_VERSION, None)
87            .await
88    }
89
90    /// Alias for [`Self::create_message_stream`] with an explicit provider name.
91    pub async fn create_anthropic_message_stream(
92        &self,
93        request: AnthropicMessageRequest,
94    ) -> Result<Pin<Box<dyn Stream<Item = Result<AnthropicMessageStreamEvent>> + Send>>> {
95        self.create_message_stream(request).await
96    }
97
98    /// Creates a streaming Messages response with explicit Anthropic headers.
99    pub async fn create_message_stream_with_headers(
100        &self,
101        mut request: AnthropicMessageRequest,
102        anthropic_version: &str,
103        anthropic_beta: Option<&str>,
104    ) -> Result<Pin<Box<dyn Stream<Item = Result<AnthropicMessageStreamEvent>> + Send>>> {
105        request.stream = Some(true);
106        validate_header(anthropic_version, "anthropic-version")?;
107        if let Some(beta) = anthropic_beta {
108            validate_header(beta, "anthropic-beta")?;
109        }
110
111        self.wait_for_slot().await;
112        let mut builder = self
113            .json_request(Method::POST, "/messages", &request)?
114            .header("anthropic-version", anthropic_version)
115            .header("accept", "text/event-stream");
116        if let Some(beta) = anthropic_beta {
117            builder = builder.header("anthropic-beta", beta);
118        }
119        let response = self.send_request(builder).await?;
120        self.handle_anthropic_message_stream_response(response)
121            .await
122    }
