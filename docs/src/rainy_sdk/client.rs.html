23fn encode_path_segment(value: &str) -> String {
24    let mut encoded = String::with_capacity(value.len());
25    for byte in value.bytes() {
26        if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
27            encoded.push(byte as char);
28        } else {
29            encoded.push('%');
30            encoded.push(char::from(b"0123456789ABCDEF"[(byte >> 4) as usize]));
31            encoded.push(char::from(b"0123456789ABCDEF"[(byte & 0x0f) as usize]));
32        }
33    }
34    encoded
35}

53/// The main client for interacting with the Rainy API.
54///
55/// `RainyClient` provides a convenient and high-level interface for making requests
56/// to the various endpoints of the Rainy API. It handles authentication, rate limiting,
57/// and retries automatically.
58///
59/// # Examples
60///
61/// ```rust,no_run
62/// use rainy_sdk::{RainyClient, Result};
63///
64/// #[tokio::main]
65/// async fn main() -> Result<()> {
66///     // Create a client using an API key from an environment variable
67///     let api_key = std::env::var("RAINY_API_KEY").expect("RAINY_API_KEY not set");
68///     let client = RainyClient::with_api_key(api_key)?;
69///
70///     // Use the client to make API calls
71///     let models = client.get_available_models().await?;
72///     println!("Available models: {:?}", models);
73///
74///     Ok(())
75/// }
76/// ```
77pub struct RainyClient {
78    /// The underlying `reqwest::Client` used for making HTTP requests.
79    client: Client,
80    /// The authentication configuration for the client.
81    auth_config: AuthConfig,
82    /// The retry configuration for handling failed requests.
83    retry_config: RetryConfig,
84
85    /// An optional rate limiter to control the request frequency.
86    /// This is only available when the `rate-limiting` feature is enabled.
87    #[cfg(feature = "rate-limiting")]
88    rate_limiter: Option<RateLimiter<NotKeyed, InMemoryState, DefaultClock>>,
89}

140    /// Creates a new `RainyClient` with a custom `AuthConfig`.
141    ///
142    /// This allows for more advanced configuration, such as setting a custom base URL or timeout.
143    ///
144    /// # Arguments
145    ///
146    /// * `auth_config` - The authentication configuration to use.
147    ///
148    /// # Returns
149    ///
150    /// A `Result` containing the new `RainyClient` or a `RainyError` if initialization fails.
151    pub fn with_config(auth_config: AuthConfig) -> Result<Self> {
152        // Validate configuration
153        auth_config.validate()?;
154
155        // Build HTTP client
156        let mut headers = HeaderMap::new();
157        headers.insert(
158            AUTHORIZATION,
159            HeaderValue::from_str(&format!("Bearer {}", auth_config.api_key.expose_secret()))
160                .map_err(|e| RainyError::Authentication {
161                    code: "INVALID_API_KEY".to_string(),
162                    message: format!("Invalid API key format: {}", e),
163                    retryable: false,
164                })?,
165        );
166        headers.insert(
167            USER_AGENT,
168            HeaderValue::from_str(&auth_config.user_agent).map_err(|e| RainyError::Network {
169                message: format!("Invalid user agent: {}", e),
170                retryable: false,
171                source_error: None,
172            })?,
173        );
174
175        let client = Client::builder()
176            .tls_backend_rustls()
177            .min_tls_version(reqwest::tls::Version::TLS_1_2)
178            // Redirects are deliberately disabled.  A configured API key or
179            // session token must never follow a redirect to another origin.
180            .redirect(reqwest::redirect::Policy::none())
181            .timeout(auth_config.timeout())
182            .default_headers(headers)
183            .build()
184            .map_err(|e| RainyError::Network {
185                message: format!("Failed to create HTTP client: {}", e),
186                retryable: false,
187                source_error: None,
188            })?;
189
190        let retry_config = RetryConfig::new(auth_config.max_retries);
191
192        #[cfg(feature = "rate-limiting")]
193        let rate_limiter = Some(RateLimiter::direct(Quota::per_second(
194            std::num::NonZeroU32::new(10).unwrap(),
195        )));
196
197        Ok(Self {
198            client,
199            auth_config,
200            retry_config,
201            #[cfg(feature = "rate-limiting")]
202            rate_limiter,
203        })
204    }

229    async fn execute_with_retry<F, Fut, T>(&self, operation: F) -> Result<T>
230    where
231        F: Fn() -> Fut,
232        Fut: Future<Output = Result<T>>,
233    {
234        self.wait_for_slot().await;
235
236        if self.auth_config.enable_retry {
237            retry_with_backoff(&self.retry_config, operation).await
238        } else {
239            operation().await
240        }
241    }

243    /// Runs a non-idempotent operation once after applying the client rate
244    /// limiter. Automatic replay of a POST can duplicate provider work or a
245    /// bill, so callers must opt into an API-level idempotency contract before
246    /// adding retries.
247    async fn execute_without_retry<F, Fut, T>(&self, operation: F) -> Result<T>
248    where
249        F: Fn() -> Fut,
250        Fut: Future<Output = Result<T>>,
251    {
252        self.wait_for_slot().await;
253        operation().await
254    }

277    pub(crate) async fn send_request(&self, request: RequestBuilder) -> Result<Response> {
278        request.send().await.map_err(|error| {
279            if error.is_timeout() {
280                RainyError::Timeout {
281                    message: "Request timed out".to_string(),
282                    duration_ms: self.auth_config.timeout_seconds.saturating_mul(1000),
283                }
284            } else {
285                RainyError::Network {
286                    message: "Failed to send request".to_string(),
287                    retryable: error.is_connect() || error.is_request(),
288                    source_error: None,
289                }
290            }
291        })
292    }

294    /// Retrieves the list of available models and providers from the API.
295    ///
296    /// # Returns
297    ///
298    /// A `Result` containing an `AvailableModels` struct on success, or a `RainyError` on failure.
299    pub async fn get_available_models(&self) -> Result<AvailableModels> {
300        #[derive(Deserialize)]
301        struct ModelListItem {
302            id: String,
303        }
304        #[derive(Deserialize)]
305        struct ModelsData {
306            data: Vec<ModelListItem>,
307        }
308        #[derive(Deserialize)]
309        struct Envelope {
310            data: ModelsData,
311        }
312
313        let url = self.api_v1_url("/models");
314
315        let operation = || async {
316            let response = self.send_request(self.client.get(&url)).await?;
317            let envelope: Envelope = self.handle_response(response).await?;
318
319            let mut providers = std::collections::HashMap::<String, Vec<String>>::new();
320            for item in envelope.data.data {
321                let provider = item
322                    .id
323                    .split_once('/')
324                    .map(|(p, _)| p.to_string())
325                    .unwrap_or_else(|| "rainy".to_string());
326                providers.entry(provider).or_default().push(item.id);
327            }
328
329            let total_models = providers.values().map(std::vec::Vec::len).sum();
330            let mut active_providers = providers.keys().cloned().collect::<Vec<_>>();
331            active_providers.sort();
332
333            Ok(AvailableModels {
334                providers,
335                total_models,
336                active_providers,
337            })
338        };
339
340        self.execute_with_retry(operation).await
341    }

474    /// Creates a streaming chat completion based on the provided request.
475    ///
476    /// # Arguments
477    ///
478    /// * `request` - A `ChatCompletionRequest` containing the model, messages, and other parameters.
479    ///
480    /// # Returns
481    ///
482    /// A `Result` containing a stream of OpenAI-style `chat.completion.chunk` events on success,
483    /// or a `RainyError` on failure.
484    pub async fn chat_completion_stream(
485        &self,
486        mut request: ChatCompletionRequest,
487    ) -> Result<Pin<Box<dyn Stream<Item = Result<ChatCompletionStreamResponse>> + Send>>> {
488        // Ensure stream is set to true
489        request.stream = Some(true);
490
491        // Retrying an inference POST can duplicate provider work or billing, so
492        // the stream connection is intentionally attempted once.
493        let operation = || async {
494            let response = self
495                .send_request(self.json_request(Method::POST, "/chat/completions", &request)?)
496                .await?;
497
498            let events = self.handle_chat_stream_response(response).await?;
499            let stream = events.filter_map(|event| async move {
500                match event {
501                    Ok(ChatStreamEvent::Chunk(chunk)) => Some(Ok(chunk)),
502                    Ok(ChatStreamEvent::Billing(_))
503                    | Ok(ChatStreamEvent::Unknown { .. })
504                    | Ok(ChatStreamEvent::Raw(_)) => None,
505                    Err(error) => Some(Err(error)),
506                }
507            });
508
509            Ok(Box::pin(stream)
510                as Pin<
511                    Box<dyn Stream<Item = Result<ChatCompletionStreamResponse>> + Send>,
512                >)
513        };
514
515        self.execute_without_retry(operation).await
516    }

665    /// Handles the HTTP response, deserializing the body into a given type `T` on success,
666    /// or mapping the error to a `RainyError` on failure.
667    ///
668    /// This is an internal method used by the various endpoint functions.
669    pub(crate) async fn handle_response<T>(&self, response: Response) -> Result<T>
670    where
671        T: serde::de::DeserializeOwned,
672    {
673        let status = response.status();
674        let request_id = response
675            .headers()
676            .get("x-request-id")
677            .and_then(|v| v.to_str().ok())
678            .map(String::from);
679        let retry_after = retry_after_seconds(response.headers());
680
681        if status.is_success() {
682            let body = read_limited_response_body(response, MAX_RESPONSE_BODY_BYTES).await?;
683            serde_json::from_slice(&body).map_err(|e| RainyError::Serialization {
684                message: format!("Failed to parse response: {}", e),
685                source_error: None,
686            })
687        } else {

695    pub(crate) async fn handle_chat_stream_response(
696        &self,
697        response: Response,
698    ) -> Result<Pin<Box<dyn Stream<Item = Result<ChatStreamEvent>> + Send>>> {
699        let status = response.status();
700        let request_id = response
701            .headers()
702            .get("x-request-id")
703            .and_then(|v| v.to_str().ok())
704            .map(String::from);
705        let retry_after = retry_after_seconds(response.headers());
706
707        if !status.is_success() {
708            let text = read_limited_response_body(response, MAX_ERROR_BODY_BYTES)
709                .await
710                .map(|body| String::from_utf8_lossy(&body).into_owned())?;
711            return self.handle_error_text(status, request_id, retry_after, text);
712        }
713
714        let stream = parse_sse_stream(response.bytes_stream()).filter_map(|event| async move {
715            match event {
716                Ok(event) if event.done || event.data.trim().is_empty() => None,
717                Ok(event) => match serde_json::from_str::<serde_json::Value>(event.data.trim()) {
718                    Ok(value) => Some(Ok(ChatStreamEvent::from_sse_event(
719                        event.event.as_deref(),
720                        value,
721                    ))),
722                    Err(_error) => Some(Err(RainyError::Serialization {
723                        message: "Failed to parse SSE JSON payload".to_string(),
724                        source_error: None,
725                    })),
726                },
727                Err(error) => Some(Err(error)),
728            }
729        });
730
731        Ok(Box::pin(stream))
732    }

810    fn handle_error_text<T>(
811        &self,
812        status: reqwest::StatusCode,
813        request_id: Option<String>,
814        retry_after: Option<u64>,
815        text: String,
816    ) -> Result<T> {
817        if let Ok(error_response) = serde_json::from_str::<ApiErrorResponse>(&text) {
818            let error = error_response.error;
819            self.map_api_error(error, status.as_u16(), request_id, retry_after)
820        } else if status == reqwest::StatusCode::TOO_MANY_REQUESTS {
821            Err(RainyError::RateLimit {
822                code: "RATE_LIMIT_EXCEEDED".to_string(),
823                message: if text.is_empty() {
824                    "Too many requests".to_string()
825                } else {
826                    text
827                },
828                retry_after,
829                current_usage: None,
830            })
831        } else {
832            Err(RainyError::Api {
833                code: status.canonical_reason().unwrap_or("UNKNOWN").to_string(),
834                message: if text.is_empty() {
835                    format!("HTTP {}", status.as_u16())
836                } else {
837                    text
838                },
839                status_code: status.as_u16(),
840                retryable: status.is_server_error(),
841                request_id,
842            })
843        }
844    }

909    /// Maps a structured API error response to a `RainyError`.
910    ///
911    /// This is an internal method.
912    fn map_api_error<T>(
913        &self,
914        error: crate::error::ApiErrorDetails,
915        status_code: u16,
916        request_id: Option<String>,
917        retry_after_header: Option<u64>,
918    ) -> Result<T> {
919        let retryable = error.retryable.unwrap_or(status_code >= 500);
920        let code = error.code.as_str();
921        let is_authentication_error = status_code == 401
922            || matches!(
923                code,
924                "INVALID_API_KEY"
925                    | "EXPIRED_API_KEY"
926                    | "UNAUTHORIZED"
927                    | "INVALID_TOKEN"
928                    | "SESSION_EXPIRED"
929                    | "SESSION_INVALID"
930            );
931        let is_access_error = status_code == 403
932            || matches!(
933                code,
934                "FORBIDDEN"
935                    | "ACCESS_DENIED"
936                    | "LAST_ADMIN_FORBIDDEN"
937                    | "ORG_ACCESS_DENIED"
938                    | "ORG_ADMIN_REQUIRED"
939                    | "USER_NOT_AUTHORIZED"
940                    | "MODEL_TIER_NOT_ALLOWED"
941                    | "MODEL_NOT_ALLOWED"
942                    | "MODEL_DISABLED_FOR_ORGANIZATION"
943                    | "MODEL_PRIVACY_POLICY_INCOMPATIBLE"
944                    | "TOOLS_NOT_ALLOWED"
945                    | "REASONING_NOT_ALLOWED"
946            );
947
948        let rainy_error = if is_authentication_error {
949            RainyError::Authentication {
950                code: error.code,
951                message: error.message,
952                retryable: false,
953            }
954        } else if is_access_error {
955            RainyError::AccessDenied {
956                code: error.code,
957                message: error.message,
958                details: error.details,
959            }
960        } else {
961            match code {
962                "REFRESH_UNSUPPORTED" => RainyError::FeatureNotAvailable {
963                    feature: "session_refresh".to_string(),
964                    message: error.message,
965                },
966                "AGENTS_ENDPOINT_REMOVED" => RainyError::FeatureNotAvailable {
967                    feature: "agents".to_string(),
968                    message: error.message,
969                },
970                "INSUFFICIENT_CREDITS" => {
971                    // Extract credit info from details if available
972                    let (current_credits, estimated_cost, reset_date) =
973                        if let Some(details) = error.details {
974                            let current = details
975                                .get("current_credits")
976                                .and_then(|v| v.as_f64())
977                                .unwrap_or(0.0);
978                            let cost = details
979                                .get("estimated_cost")
980                                .and_then(|v| v.as_f64())
981                                .unwrap_or(0.0);
982                            let reset = details
983                                .get("reset_date")
984                                .and_then(|v| v.as_str())
985                                .map(String::from);
986                            (current, cost, reset)
987                        } else {
988                            (0.0, 0.0, None)
989                        };
990
991                    RainyError::InsufficientCredits {
992                        code: error.code,
993                        message: error.message,
994                        current_credits,
995                        estimated_cost,
996                        reset_date,
997                    }
998                }
999                "RATE_LIMIT_EXCEEDED" => {
1000                    let retry_after = error
1001                        .details
1002                        .as_ref()
1003                        .and_then(|d| d.get("retry_after"))
1004                        .and_then(|v| v.as_u64())
1005                        .or(retry_after_header);
1006
1007                    RainyError::RateLimit {
1008                        code: error.code,
1009                        message: error.message,
1010                        retry_after,
1011                        current_usage: None,
1012                    }
1013                }
1014                "INVALID_REQUEST" | "MISSING_REQUIRED_FIELD" | "INVALID_MODEL" => {
1015                    RainyError::InvalidRequest {
1016                        code: error.code,
1017                        message: error.message,
1018                        details: error.details,
1019                    }
1020                }
1021                "PROVIDER_ERROR" | "PROVIDER_UNAVAILABLE" => {
1022                    let provider = error
1023                        .details
1024                        .as_ref()
1025                        .and_then(|d| d.get("provider"))
1026                        .and_then(|v| v.as_str())
1027                        .unwrap_or("unknown")
1028                        .to_string();
1029
1030                    RainyError::Provider {
1031                        code: error.code,
1032                        message: error.message,
1033                        provider,
1034                        retryable,
1035                    }
1036                }
1037                _ if status_code == 429 => RainyError::RateLimit {
1038                    code: error.code,
1039                    message: error.message,
1040                    retry_after: retry_after_header,
1041                    current_usage: error
1042                        .details
1043                        .as_ref()
1044                        .and_then(|details| details.get("current_usage"))
1045                        .and_then(|value| value.as_str())
1046                        .map(String::from),
1047                },
1048                _ => RainyError::Api {
1049                    code: error.code,
1050                    message: error.message,
1051                    status_code,
1052                    retryable,
1053                    request_id: request_id.clone(),
1054                },
1055            }
1056        };
1057
1058        Err(rainy_error)
1059    }

1108    /// Makes a generic HTTP request to the API.
1109    ///
1110    /// This is an internal method kept for compatibility with endpoint implementations.
1111    pub(crate) async fn make_request<T: serde::de::DeserializeOwned>(
1112        &self,
1113        method: reqwest::Method,
1114        endpoint: &str,
1115        body: Option<serde_json::Value>,
1116    ) -> Result<T> {
1117        self.wait_for_slot().await;
1118        let request = match body {
1119            Some(body) => self.json_request(method, endpoint, &body)?,
1120            None => self.api_request(method, endpoint),
1121        };
1122
1123        let response = self.send_request(request).await?;
1124        self.handle_response(response).await
1125    }
