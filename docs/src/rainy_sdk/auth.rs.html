22/// Authentication convention used by a protocol request.
23#[derive(Debug, Clone, Copy, PartialEq, Eq)]
24pub(crate) enum AuthScheme {
25    /// `Authorization: Bearer <key>`.
26    Bearer,
27    /// `x-api-key: <key>`.
28    XApiKey,
29}

31/// Configuration for API-key authenticated protocol clients.
32///
33/// The API key is held as a [`SecretString`] and is never included in the
34/// configuration's `Debug` or `Display` output. Key format validation is
35/// intentionally protocol-neutral so the same client can target Rainy or a
36/// compatible OpenAI/Anthropic endpoint.
37#[derive(Clone)]
38pub struct AuthConfig {
39    /// Secret used for protocol authentication.
40    pub api_key: SecretString,
41    /// Host base URL used by root-level routes.
42    pub base_url: String,
43    /// Explicit versioned API base URL. If unset, `/api/v1` is appended to the
44    /// base URL unless the base URL already has a non-root path.
45    pub api_base_url: Option<String>,
46    /// Request timeout in seconds.
47    pub timeout_seconds: u64,
48    /// Maximum retries for explicitly safe operations.
49    pub max_retries: u32,
50    /// Whether safe operations may retry.
51    pub enable_retry: bool,
52    /// User-Agent header.
53    pub user_agent: String,
54}

106    /// Validates credentials, URLs, and header-safe configuration.
107    pub fn validate(&self) -> Result<()> {
108        let key = self.api_key.expose_secret();
109        if key.is_empty() {
110            return Err(RainyError::Authentication {
111                code: "EMPTY_API_KEY".to_string(),
112                message: "API key cannot be empty".to_string(),
113                retryable: false,
114            });
115        }
116        if key.len() > 4096
117            || key
118                .chars()
119                .any(|character| character.is_whitespace() || character.is_control())
120        {
121            return Err(RainyError::Authentication {
122                code: "INVALID_API_KEY".to_string(),
123                message: "API key contains invalid characters".to_string(),
124                retryable: false,
125            });
126        }
127        validate_service_url(&self.base_url, "INVALID_BASE_URL")?;
128        if let Some(api_base_url) = &self.api_base_url {
129            validate_service_url(api_base_url, "INVALID_API_BASE_URL")?;
130        }
131        HeaderValue::from_str(&self.user_agent).map_err(|_| RainyError::InvalidRequest {
132            code: "INVALID_USER_AGENT".to_string(),
133            message: "User-Agent contains invalid header characters".to_string(),
134            details: None,
135        })?;
136        Ok(())
137    }

139    /// Builds Bearer headers for OpenAI-compatible requests.
140    pub fn build_headers(&self) -> Result<HeaderMap> {
141        self.validate()?;
142        self.build_protocol_headers(AuthScheme::Bearer, None)
143    }

151    pub(crate) fn build_protocol_headers(
152        &self,
153        scheme: AuthScheme,
154        anthropic_version: Option<&str>,
155    ) -> Result<HeaderMap> {
156        let mut headers = HeaderMap::new();
157        headers.insert(
158            USER_AGENT,
159            HeaderValue::from_str(&self.user_agent).map_err(|_| RainyError::InvalidRequest {
160                code: "INVALID_USER_AGENT".to_string(),
161                message: "User-Agent contains invalid header characters".to_string(),
162                details: None,
163            })?,
164        );
165        headers.insert(CONTENT_TYPE, HeaderValue::from_static("application/json"));
166
167        match scheme {
168            AuthScheme::Bearer => {
169                let value = format!("Bearer {}", self.api_key.expose_secret());
170                headers.insert(
171                    AUTHORIZATION,
172                    HeaderValue::from_str(&value).map_err(|_| RainyError::Authentication {
173                        code: "INVALID_API_KEY".to_string(),
174                        message: "API key cannot be used in an HTTP header".to_string(),
175                        retryable: false,
176                    })?,
177                );
178            }
179            AuthScheme::XApiKey => {
180                headers.insert(
181                    "x-api-key",
182                    HeaderValue::from_str(self.api_key.expose_secret()).map_err(|_| {
183                        RainyError::Authentication {
184                            code: "INVALID_API_KEY".to_string(),
185                            message: "API key cannot be used in an HTTP header".to_string(),
186                            retryable: false,
187                        }
188                    })?,
189                );
190                let version = anthropic_version.ok_or_else(|| RainyError::InvalidRequest {
191                    code: "MISSING_ANTHROPIC_VERSION".to_string(),
192                    message: "Anthropic requests require an API version".to_string(),
193                    details: None,
194                })?;
195                headers.insert(
196                    "anthropic-version",
197                    HeaderValue::from_str(version).map_err(|_| RainyError::InvalidRequest {
198                        code: "INVALID_ANTHROPIC_VERSION".to_string(),
199                        message: "Anthropic API version is not a valid header value".to_string(),
200                        details: None,
201                    })?,
202                );
203            }
204        }
205        Ok(headers)
206    }
207
208    /// Returns the configured timeout.
209    pub fn timeout(&self) -> Duration {
210        Duration::from_secs(self.timeout_seconds)
211    }

244/// Returns a host-only URL label suitable for diagnostics.
245pub(crate) fn safe_url_label(value: &str) -> String {
246    let Ok(url) = Url::parse(value) else {
247        return "<invalid-url>".to_string();
248    };
249    match url.host_str() {
250        Some(host) => format!("{}://{}", url.scheme(), host),
251        None => "<invalid-url>".to_string(),
252    }
253}

255/// Validates a configured service URL without logging or normalizing secrets.
256pub(crate) fn validate_service_url(value: &str, code: &str) -> Result<Url> {
257    if value.trim() != value || value.is_empty() {
258        return Err(RainyError::InvalidRequest {
259            code: code.to_string(),
260            message: "service URL must not be empty or surrounded by whitespace".to_string(),
261            details: None,
262        });
263    }
264    let url = Url::parse(value).map_err(|_| RainyError::InvalidRequest {
265        code: code.to_string(),
266        message: "service URL is not valid".to_string(),
267        details: None,
268    })?;
269    let host = url.host_str().ok_or_else(|| RainyError::InvalidRequest {
270        code: code.to_string(),
271        message: "service URL must include a host".to_string(),
272        details: None,
273    })?;
274    if url.username() != ""
275        || url.password().is_some()
276        || url.query().is_some()
277        || url.fragment().is_some()
278    {
279        return Err(RainyError::InvalidRequest {
280            code: code.to_string(),
281            message: "service URL must not contain credentials, query parameters, or fragments"
282                .to_string(),
283            details: None,
284        });
285    }
286
287    match url.scheme() {
288        "https" => {}
289        "http" if is_loopback_host(host) => {}
290        _ => {
291            return Err(RainyError::InvalidRequest {
292                code: code.to_string(),
293                message: "service URL must use HTTPS; HTTP is allowed only for loopback testing"
294                    .to_string(),
295                details: None,
296            });
297        }
298    }
299    Ok(url)
300}

302fn is_loopback_host(host: &str) -> bool {
303    host.eq_ignore_ascii_case("localhost")
304        || host
305            .parse::<std::net::IpAddr>()
306            .is_ok_and(|address| address.is_loopback())
307}

309/// Serializes a request body while enforcing a byte limit before sending it.
310pub(crate) fn serialize_json_body<T: serde::Serialize>(body: &T, limit: usize) -> Result<Vec<u8>> {
311    let bytes = serde_json::to_vec(body).map_err(|error| RainyError::Serialization {
312        message: "failed to serialize JSON request".to_string(),
313        source_error: Some(error.to_string()),
314    })?;
315    if bytes.len() > limit {
316        return Err(RainyError::PayloadTooLarge {
317            message: "JSON request body exceeds the configured safety limit".to_string(),
318            max_bytes: limit,
319        });
320    }
321    Ok(bytes)
322}

324/// Reads a response body into bounded memory.
325pub(crate) async fn read_limited_response_body(
326    response: Response,
327    limit: usize,
328) -> Result<Vec<u8>> {
329    if response
330        .content_length()
331        .is_some_and(|length| length > limit as u64)
332    {
333        return Err(RainyError::PayloadTooLarge {
334            message: "response body exceeds the configured safety limit".to_string(),
335            max_bytes: limit,
336        });
337    }
338
339    let mut body = Vec::new();
340    let mut stream = response.bytes_stream();
341    while let Some(chunk) = stream.next().await {
342        let chunk = chunk.map_err(RainyError::from)?;
343        if body.len().saturating_add(chunk.len()) > limit {
344            return Err(RainyError::PayloadTooLarge {
345                message: "response body exceeds the configured safety limit".to_string(),
346                max_bytes: limit,
347            });
348        }
349        body.extend_from_slice(&chunk);
350    }
351    Ok(body)
352}

354/// Parses a numeric `Retry-After` header and caps it to one day.
355pub(crate) fn retry_after_seconds(headers: &reqwest::header::HeaderMap) -> Option<u64> {
356    headers
357        .get("retry-after")
358        .and_then(|value| value.to_str().ok())
359        .and_then(|value| value.trim().parse::<u64>().ok())
360        .map(|value| value.min(MAX_RETRY_AFTER_SECONDS))
361}

363/// A simple rate limiter retained only for source compatibility.
364#[deprecated(note = "Use the optional governor-based rate limiting in RainyClient")]
365#[derive(Debug)]
366pub struct RateLimiter {
367    requests_per_minute: u32,
368    last_request: std::time::Instant,
369    request_count: u32,
370}
371
372#[allow(deprecated)]
373impl RateLimiter {
374    /// Creates a limiter.
375    pub fn new(requests_per_minute: u32) -> Self {
376        Self {
377            requests_per_minute,
378            last_request: std::time::Instant::now(),
379            request_count: 0,
380        }
381    }
382
383    /// Waits until the next request may be sent.
384    pub async fn wait_if_needed(&mut self) -> Result<()> {
385        let now = std::time::Instant::now();
386        let elapsed = now.duration_since(self.last_request);
387        if elapsed >= Duration::from_secs(60) {
388            self.request_count = 0;
389            self.last_request = now;
390        }
391        if self.request_count >= self.requests_per_minute {
392            tokio::time::sleep(Duration::from_secs(60) - elapsed).await;
393            self.request_count = 0;
394            self.last_request = std::time::Instant::now();
395        }
396        self.request_count = self.request_count.saturating_add(1);
397        Ok(())
398    }
