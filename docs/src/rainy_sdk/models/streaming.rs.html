43/// A typed Chat streaming event.
44#[derive(Debug, Clone, PartialEq)]
45pub enum ChatStreamEvent {
46    /// Standard OpenAI-compatible chunk.
47    Chunk(ChatCompletionStreamResponse),
48    /// Rainy native billing event.
49    Billing(RainyBillingStreamEvent),
50    /// A named event not yet classified by this SDK.
51    Unknown {
52        /// Native SSE event name.
53        event: String,
54        /// Parsed event payload.
55        data: Value,
56    },
57    /// An unnamed payload that did not match a known chunk.
58    Raw(Value),
59}
60
61impl ChatStreamEvent {
62    /// Classifies an unnamed JSON payload.
63    pub fn from_value(value: Value) -> Self {
64        if let Ok(chunk) = serde_json::from_value::<ChatCompletionStreamResponse>(value.clone()) {
65            return Self::Chunk(chunk);
66        }
67        if let Ok(billing) = serde_json::from_value::<RainyBillingStreamEvent>(value.clone())
68            && (billing.plan_id.is_some()
69                || billing.charged_credits.is_some()
70                || billing.usage.is_some())
71        {
72            return Self::Billing(billing);
73        }
74        Self::Raw(value)
75    }
76
77    /// Classifies a payload using the native event name before shape fallback.
78    pub fn from_sse_event(event_name: Option<&str>, value: Value) -> Self {
79        let Some(event_name) = event_name.filter(|name| !name.is_empty()) else {
80            return Self::from_value(value);
81        };
82
83        if event_name.eq_ignore_ascii_case("rainy.billing")
84            && let Ok(billing) = serde_json::from_value::<RainyBillingStreamEvent>(value.clone())
85            && (billing.plan_id.is_some()
86                || billing.charged_credits.is_some()
87                || billing.usage.is_some())
88        {
89            return Self::Billing(billing);
90        }
91
92        if event_name.eq_ignore_ascii_case("message")
93            || event_name.eq_ignore_ascii_case("chat.completion.chunk")
94        {
95            return match serde_json::from_value::<ChatCompletionStreamResponse>(value.clone()) {
96                Ok(chunk) => Self::Chunk(chunk),
97                Err(_) => Self::Unknown {
98                    event: event_name.to_string(),
99                    data: value,
100                },
101            };
102        }
103
104        Self::Unknown {
105            event: event_name.to_string(),
106            data: value,
107        }
108    }
109}

111/// Standard OpenAI-compatible Chat completion chunk.
112#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
113pub struct ChatCompletionStreamResponse {
114    /// Completion identifier.
115    pub id: String,
116    /// Object label.
117    pub object: String,
118    /// Creation timestamp.
119    pub created: u64,
120    /// Model identifier.
121    pub model: String,
122    /// Stream choices.
123    pub choices: Vec<ChatCompletionStreamChoice>,
124    /// Optional terminal usage.
125    #[serde(default, skip_serializing_if = "Option::is_none")]
126    pub usage: Option<Usage>,
127    /// Future-compatible chunk fields.
128    #[serde(flatten, default, skip_serializing_if = "BTreeMap::is_empty")]
129    pub extra: BTreeMap<String, Value>,
130}
