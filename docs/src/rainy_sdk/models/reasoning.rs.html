1//! Typed reasoning controls shared by Chat, Responses, and model discovery.
2//!
3//! Rainy deliberately forwards the provider-neutral OpenRouter reasoning
4//! contract. The SDK therefore models only controls that are part of that
5//! contract (`enabled`, `effort`, `max_tokens`, and `exclude`). Provider or
6//! model-specific controls must be carried in the explicit extension map and
7//! must be selected from the model catalog rather than inferred from a model
8//! name.
9
10use serde::{Deserialize, Deserializer, Serialize, Serializer};
11use serde_json::Value;
12use std::collections::BTreeMap;
13use std::fmt;

15/// Canonical reasoning-effort values accepted by Rainy.
16#[derive(Debug, Clone, PartialEq, Eq, Hash)]
17pub enum ReasoningEffort {
18    /// Disable reasoning when the model supports an explicit `none` value.
19    None,
20    /// Request the smallest supported reasoning effort.
21    Minimal,
22    /// Request low reasoning effort.
23    Low,
24    /// Request medium reasoning effort.
25    Medium,
26    /// Request high reasoning effort.
27    High,
28    /// Request extra-high reasoning effort.
29    XHigh,
30    /// Request the maximum reasoning effort.
31    Max,
32    /// Preserve a future catalog/provider value without silently remapping it.
33    Custom(String),
34}
35
36impl ReasoningEffort {
37    /// Returns the exact wire value for this effort.
38    pub fn as_str(&self) -> &str {
39        match self {
40            Self::None => "none",
41            Self::Minimal => "minimal",
42            Self::Low => "low",
43            Self::Medium => "medium",
44            Self::High => "high",
45            Self::XHigh => "xhigh",
46            Self::Max => "max",
47            Self::Custom(value) => value.as_str(),
48        }
49    }
50
51    /// Returns whether the value is one of Rainy's canonical effort levels.
52    pub fn is_canonical(&self) -> bool {
53        !matches!(self, Self::Custom(_))
54    }
55
56    /// Parses an effort value while preserving unknown values.
57    pub fn parse(value: impl AsRef<str>) -> Self {
58        match value.as_ref() {
59            "none" => Self::None,
60            "minimal" => Self::Minimal,
61            "low" => Self::Low,
62            "medium" => Self::Medium,
63            "high" => Self::High,
64            "xhigh" => Self::XHigh,
65            "max" => Self::Max,
66            other => Self::Custom(other.to_string()),
67        }
68    }
69}
70
71impl fmt::Display for ReasoningEffort {
72    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
73        formatter.write_str(self.as_str())
74    }
75}

77impl From<&str> for ReasoningEffort {
78    fn from(value: &str) -> Self {

80    }
81}

108/// A typed reasoning configuration for Rainy Chat and Responses requests.
109#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Default)]
110pub struct ReasoningConfig {
111    /// Explicitly enable or disable reasoning when the model exposes a toggle.
112    #[serde(skip_serializing_if = "Option::is_none")]
113    pub enabled: Option<bool>,
114    /// Canonical or catalog-declared effort value.
115    #[serde(skip_serializing_if = "Option::is_none")]
116    pub effort: Option<ReasoningEffort>,
117    /// Explicit reasoning-token budget when the selected model declares this control.
118    #[serde(skip_serializing_if = "Option::is_none")]
119    pub max_tokens: Option<u32>,
120    /// Exclude reasoning from provider output/processing when supported.
121    #[serde(skip_serializing_if = "Option::is_none")]
122    pub exclude: Option<bool>,
123    /// Explicit provider/model extension fields inside the reasoning object.
124    #[serde(flatten, default, skip_serializing_if = "BTreeMap::is_empty")]
125    pub extra: BTreeMap<String, Value>,
126}
127
128impl ReasoningConfig {
129    /// Creates an empty configuration that enables the model's adaptive/default reasoning.
130    pub fn adaptive() -> Self {
131        Self {
132            enabled: Some(true),
133            ..Self::default()
134        }
135    }
136
137    /// Creates an effort-based reasoning configuration.
138    pub fn effort(effort: impl Into<ReasoningEffort>) -> Self {
139        Self {
140            effort: Some(effort.into()),
141            ..Self::default()
142        }
143    }
144
145    /// Creates a manual numeric reasoning-budget configuration.
146    pub fn manual_budget(max_tokens: u32) -> Self {
147        Self {
148            max_tokens: Some(max_tokens),
149            ..Self::default()
150        }
151    }
152
153    /// Creates an exclusion configuration.
154    pub fn excluded() -> Self {
155        Self {
156            exclude: Some(true),
157            ..Self::default()
158        }
159    }
160
161    /// Sets the enabled flag.
162    pub fn with_enabled(mut self, enabled: bool) -> Self {
163        self.enabled = Some(enabled);
164        self
165    }
166
167    /// Sets the effort control.
168    pub fn with_effort(mut self, effort: impl Into<ReasoningEffort>) -> Self {
169        self.effort = Some(effort.into());
170        self
171    }
172
173    /// Sets the numeric reasoning-token budget.
174    pub fn with_max_tokens(mut self, max_tokens: u32) -> Self {
175        self.max_tokens = Some(max_tokens);
176        self
177    }
178
179    /// Sets the exclusion flag.
180    pub fn with_exclude(mut self, exclude: bool) -> Self {
181        self.exclude = Some(exclude);
182        self
183    }
184
185    /// Adds an explicit extension field to the reasoning object.
186    pub fn with_extra(mut self, key: impl Into<String>, value: Value) -> Self {
187        self.extra.insert(key.into(), value);
188        self
189    }
190
191    /// Checks for controls that are mutually exclusive under Rainy's contract.
192    pub fn validate(&self) -> Result<(), String> {
193        if self.effort.is_some() && self.max_tokens.is_some() {
194            return Err(
195                "reasoning.effort and reasoning.max_tokens cannot be sent together".to_string(),
196            );
197        }
198        Ok(())
199    }
200}

202/// Chat/Responses reasoning payload, including the legacy boolean form.
203#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
204#[serde(untagged)]
205pub enum ReasoningRequest {
206    /// Boolean reasoning toggle accepted by Chat compatibility routes.
207    Enabled(bool),
208    /// Structured Rainy/OpenRouter reasoning controls.
209    Config(ReasoningConfig),
210    /// A deliberately preserved compatibility value for callers migrating from v0.x.
211    /// New code should use [`ReasoningConfig`] instead.
212    Raw(Value),
213}
214
215impl From<bool> for ReasoningRequest {
216    fn from(value: bool) -> Self {
217        Self::Enabled(value)
218    }
219}
220
221impl From<ReasoningConfig> for ReasoningRequest {
222    fn from(value: ReasoningConfig) -> Self {
223        Self::Config(value)
224    }
225}
226
227impl From<ReasoningEffort> for ReasoningRequest {
228    fn from(value: ReasoningEffort) -> Self {
229        Self::Config(ReasoningConfig::effort(value))
230    }
231}
232
233impl From<Value> for ReasoningRequest {
234    fn from(value: Value) -> Self {
235        match value {
236            Value::Bool(value) => Self::Enabled(value),
237            Value::Object(object) => {
238                let original = Value::Object(object.clone());
239                serde_json::from_value(original)
240                    .map(Self::Config)
241                    .unwrap_or(Self::Raw(Value::Object(object)))
242            }
243            other => Self::Raw(other),
244        }
245    }
246}

248/// High-level reasoning intent used when building a request from capabilities.
249#[derive(Debug, Clone, PartialEq, Eq)]
250pub enum ReasoningControl {
251    /// Let the selected model choose its adaptive/default reasoning behavior.
252    Adaptive,
253    /// Use an explicit effort value declared by the selected model.
254    Effort(ReasoningEffort),
255    /// Use an explicit numeric budget declared by the selected model.
256    ManualBudget(u32),
257    /// Ask the selected model to exclude reasoning where supported.
258    Disabled,
259}
260
261impl ReasoningControl {
262    /// Converts this intent into the provider-neutral Rainy reasoning object.
263    pub fn as_config(&self) -> ReasoningConfig {
264        match self {
265            Self::Adaptive => ReasoningConfig::adaptive(),
266            Self::Effort(effort) => ReasoningConfig::effort(effort.clone()),
267            Self::ManualBudget(max_tokens) => ReasoningConfig::manual_budget(*max_tokens),
268            Self::Disabled => ReasoningConfig::excluded(),
269        }
270    }
271}

273/// A model-declared numeric reasoning-budget choice.
274#[derive(Debug, Clone, Copy, PartialEq, Eq)]
275pub enum ReasoningBudget {
276    /// A concrete number of reasoning tokens.
277    Tokens(u32),
278    /// The model's catalog-declared dynamic budget value.
279    Dynamic,
280    /// The model's catalog-declared disabled value.
281    Disabled,
282}
283
284impl ReasoningBudget {
285    /// Creates a concrete token budget.
286    pub const fn tokens(value: u32) -> Self {
287        Self::Tokens(value)
288    }
289
290    /// Returns the concrete value, if this choice is already numeric.
291    pub const fn as_tokens(self) -> Option<u32> {
292        match self {
293            Self::Tokens(value) => Some(value),
294            Self::Dynamic | Self::Disabled => None,
295        }
296    }
297}

299#[cfg(test)]
300mod tests {
301    use super::*;
302
303    #[test]
304    fn serializes_canonical_effort_and_manual_budget_without_remapping() {
305        let effort = serde_json::to_value(ReasoningConfig::effort(ReasoningEffort::XHigh))
306            .expect("effort JSON");
307        assert_eq!(effort["effort"], "xhigh");
308
309        let budget =
310            serde_json::to_value(ReasoningConfig::manual_budget(2048)).expect("budget JSON");
311        assert_eq!(budget["max_tokens"], 2048);
312
313        let custom = serde_json::to_value(ReasoningEffort::Custom("provider_next".to_string()))
314            .expect("custom JSON");
315        assert_eq!(custom, "provider_next");
316    }
317
318    #[test]
319    fn rejects_conflicting_effort_and_budget() {
320        let config = ReasoningConfig::effort(ReasoningEffort::High).with_max_tokens(1024);
321        assert!(config.validate().is_err());
322    }
